import requests
BASE = "https://api.agenttalk-host"
KEY = "ak_..."

def search_services(query, limit=5):
    r = requests.post(f"{BASE}/search", headers={"Authorization": f"Bearer {KEY}"},
                      json={"query": query, "limit": limit},)
    r.raise_for_status()
    return r.json()["results"]

def invoke_service(service_id, payload, agent_id="autogen"):
    r = requests.post(f"{BASE}/calls/invoke",
        headers={"Authorization": f"Bearer {KEY}"},
        json={"service_id": service_id, "agent_id": agent_id,
              "request_payload": payload})
    r.raise_for_status()
    return r.json()
