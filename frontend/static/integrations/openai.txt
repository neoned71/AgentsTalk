// OpenAI Agents / Chat Completions function calling
const base = process.env.AGENTTALK_BASE;
const key = process.env.AGENTTALK_KEY;

async function agenttalkFetch(path, payload) {
  const r = await fetch(`${base}${path}`, {
    method: "POST",
    headers: { "Authorization": `Bearer ${key}`, "Content-Type": "application/json" },
    body: JSON.stringify(payload)
  });
  if (r.status === 402) throw new Error(JSON.stringify(await r.json()));
  return r.json();
}
